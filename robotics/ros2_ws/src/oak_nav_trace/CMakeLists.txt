cmake_minimum_required(VERSION 3.16)
project(oak_nav_trace LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(nav2_controller REQUIRED)
find_package(nav2_core REQUIRED)
find_package(pluginlib REQUIRED)
find_package(std_msgs REQUIRED)
add_library(oak_progress_trace SHARED src/progress_trace.cpp)
ament_target_dependencies(oak_progress_trace nav2_controller nav2_core pluginlib std_msgs)
target_compile_options(oak_progress_trace PRIVATE -Wall -Wextra -Wpedantic -Werror)
pluginlib_export_plugin_description_file(nav2_core plugins.xml)
install(TARGETS oak_progress_trace LIBRARY DESTINATION lib)
if(BUILD_TESTING)
  find_package(rclcpp REQUIRED)
  find_package(std_srvs REQUIRED)
  find_package(rmw_fastrtps_shared_cpp REQUIRED)
  find_package(fastrtps REQUIRED)
  add_executable(oak_service_qos_probe src/service_qos_probe.cpp)
  ament_target_dependencies(oak_service_qos_probe rclcpp std_srvs rmw_fastrtps_shared_cpp)
  target_link_libraries(oak_service_qos_probe fastrtps)
  install(TARGETS oak_service_qos_probe DESTINATION lib/${PROJECT_NAME})
  enable_testing()
  add_test(NAME bounded_service_discovery COMMAND oak_service_qos_probe)
  get_filename_component(OAK_REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
  set_tests_properties(bounded_service_discovery PROPERTIES ENVIRONMENT
    "RMW_IMPLEMENTATION=rmw_fastrtps_cpp;RMW_FASTRTPS_USE_QOS_FROM_XML=0;FASTRTPS_DEFAULT_PROFILES_FILE=${OAK_REPO}/configs/robotics/phase32_fastdds_services.xml;ROS_DOMAIN_ID=85;ROS_LOCALHOST_ONLY=1")
endif()
ament_package()
