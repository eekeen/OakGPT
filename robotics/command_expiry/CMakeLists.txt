cmake_minimum_required(VERSION 3.16)
project(oak_command_expiry LANGUAGES CXX)
option(OAK_BUILD_GAZEBO_PLUGIN "Build the Gazebo Harmonic system plugin" OFF)
option(OAK_SANITIZERS "Enable address/undefined sanitizers" OFF)
add_library(expiry_rules INTERFACE)
target_include_directories(expiry_rules INTERFACE include)
target_compile_features(expiry_rules INTERFACE cxx_std_17)
if(MSVC)
  target_compile_options(expiry_rules INTERFACE /W4 /WX)
else()
  target_compile_options(expiry_rules INTERFACE -Wall -Wextra -Wpedantic -Werror)
  if(OAK_SANITIZERS)
    target_compile_options(expiry_rules INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(expiry_rules INTERFACE -fsanitize=address,undefined)
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(expiry-tests tests/expiry_test.cpp)
  target_link_libraries(expiry-tests PRIVATE expiry_rules)
  add_test(NAME expiry_state_and_properties COMMAND expiry-tests)
endif()
if(OAK_BUILD_GAZEBO_PLUGIN)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Plugin uses Linux CLOCK_BOOTTIME; portable core tests remain available")
  endif()
  find_package(gz-sim8 REQUIRED)
  find_package(gz-plugin2 REQUIRED COMPONENTS register)
  find_package(gz-transport13 REQUIRED)
  add_library(oak_command_expiry SHARED src/gazebo_expiry.cpp)
  target_link_libraries(oak_command_expiry PRIVATE expiry_rules gz-sim8::gz-sim8
    gz-plugin2::gz-plugin2 gz-transport13::gz-transport13)
  install(TARGETS oak_command_expiry LIBRARY DESTINATION lib)
  if(BUILD_TESTING)
    add_executable(expiry-message-tests tests/gazebo_message_test.cpp)
    target_link_libraries(expiry-message-tests PRIVATE expiry_rules gz-sim8::gz-sim8)
    add_test(NAME gazebo_message_boundaries COMMAND expiry-message-tests)
  endif()
endif()
