cmake_minimum_required(VERSION 3.16)
project(oak_safety VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(OAK_SANITIZERS "Enable address/undefined sanitizers on GCC/Clang" OFF)
add_library(oak_safety src/validator.cpp)
target_include_directories(oak_safety PUBLIC include third_party)
if(MSVC)
  target_compile_options(oak_safety PRIVATE /W4 /WX)
else()
  target_compile_options(oak_safety PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(OAK_SANITIZERS)
    target_compile_options(oak_safety PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(oak_safety PUBLIC -fsanitize=address,undefined)
  endif()
endif()
add_executable(oak-validate src/main.cpp)
target_link_libraries(oak-validate PRIVATE oak_safety)
include(CTest)
if(BUILD_TESTING)
  add_executable(oak-safety-tests tests/test_validator.cpp)
  target_link_libraries(oak-safety-tests PRIVATE oak_safety)
  add_test(NAME unit_and_mutation_properties COMMAND oak-safety-tests)
endif()
install(TARGETS oak-validate RUNTIME DESTINATION bin)
